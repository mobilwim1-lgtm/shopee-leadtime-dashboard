FUNGSI GET: Untuk menarik data ke Dashboard HTML function doGet() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheetLogs = ss.getSheetByName("logs"); const dataLogs = sheetLogs ? getSheetDataWithHeader(sheetLogs, true) : []; const sheetUnits = ss.getSheetByName("units"); const dataUnits = sheetUnits ? getSheetDataWithHeader(sheetUnits, false) : []; const sheetWh = ss.getSheetByName("warehouses"); const dataWh = sheetWh ? getSheetDataWithHeader(sheetWh, false) : []; const result = { logs: dataLogs, units: dataUnits, warehouses: dataWh }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 2. FUNGSI POST: Untuk menyimpan/mengubah data dari Dashboard HTML function doPost(e) { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheetLogs = ss.getSheetByName("logs"); if (!sheetLogs) throw new Error("Sheet dengan nama 'logs' tidak ditemukan!"); const payload = JSON.parse(e.postData.contents); // Aksi EDIT/UPDATE Data (Timestamp Inline atau STATUS 2) if (payload.action === "update") { const rowIndex = parseInt(payload.ROW_INDEX); const headers = sheetLogs.getRange(1, 1, 1, sheetLogs.getLastColumn()).getValues()[0]; for (let key in payload) { if (key !== "action" && key !== "ROW_INDEX" && key !== "NOPOL") { let colIndex = headers.indexOf(key) + 1; if (colIndex > 0) { sheetLogs.getCell(rowIndex, colIndex).setValue(payload[key]); } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data berhasil diperbarui!" })) .setMimeType(ContentService.MimeType.JSON); } // Aksi INPUT DATA BARU else { const headers = sheetLogs.getRange(1, 1, 1, sheetLogs.getLastColumn()).getValues()[0]; const newRow = new Array(headers.length).fill(""); headers.forEach((header, index) => { if (payload[header] !== undefined) { newRow[index] = payload[header]; } }); sheetLogs.appendRow(newRow); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data baru berhasil disimpan!" })) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message })) .setMimeType(ContentService.MimeType.JSON); } } // Fungsi Pembantu Formating Data JSON dari Sheet function getSheetDataWithHeader(sheet, includeRowIndex) { const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow <= 1) return []; const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; const values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); return values.map((row, idx) => { let obj = {}; if (includeRowIndex) obj["ROW_INDEX"] = idx + 2; // Menyimpan info indeks baris asli di Sheet headers.forEach((header, colIdx) => { if (header) { let val = row[colIdx]; if (val instanceof Date) { if (header.includes("DATE")) { obj[header] = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } else { obj[header] = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } } else { obj[header] = val; } } }); return obj; }); } Deploy Web App: Klik tombol Terapkan (Deploy) di kanan atas > Pilih Penerapan baru (New deployment).
